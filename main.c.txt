#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include "rezervacija.h"

int main() {
    int izbor;
    ucitajRezervacije(&rezervacije, &brojRezervacija);

    do {
        printf("\n--- HOTELSKI MENI ---\n");
        printf("1. Dodaj rezervaciju\n");
        printf("2. Prikazi rezervacije\n");
        printf("3. Azuriraj rezervaciju\n");
        printf("4. Obrisi rezervaciju\n");
        printf("5. Sortiraj po nocenjima\n");
        printf("6. Pretrazi po broju sobe\n");
        printf("7. Izlaz\n");
        printf("Izbor: ");
        scanf("%d", &izbor);

        switch ((MeniOpcije)izbor) {
        case DODAJ_REZERVACIJU:
            dodajRezervaciju(&rezervacije, &brojRezervacija);
            break;
        case PRIKAZI_REZERVACIJE:
            prikaziRezervacije(rezervacije, brojRezervacija);
            break;
        case AZURIRAJ_REZERVACIJU:
            azurirajRezervaciju(rezervacije, brojRezervacija);
            break;
        case OBRISI_REZERVACIJU:
            obrisiRezervaciju(rezervacije, &brojRezervacija);
            break;
        case SORTIRAJ_PO_NOCENJIMA:
            sortirajPoNocenjima(rezervacije, brojRezervacija);
            break;
        case PRETRAZI_PO_SOBI: {
            int brojSobe;
            printf("Unesite broj sobe: ");
            scanf("%d", &brojSobe);
            pretraziPoSobi(rezervacije, brojRezervacija, brojSobe);
            break;
        }
        case IZLAZ:
            spremiRezervacije(rezervacije, brojRezervacija);
            ocistiMemoriju(&rezervacije);
            printf("Izlaz iz programa.\n");
            break;
        default:
            printf("Nepoznata opcija.\n");
        }
    } while (izbor != IZLAZ);

    return 0;
}
